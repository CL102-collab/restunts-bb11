cmake_minimum_required(VERSION 3.25)
project(restunts_mt32emu_rebuild LANGUAGES C CXX)
if(EMSCRIPTEN)
    set(libmt32emu_SHARED OFF CACHE BOOL "" FORCE)
else()
    set(libmt32emu_SHARED ON CACHE BOOL "" FORCE)
endif()
set(libmt32emu_C_INTERFACE ON CACHE BOOL "" FORCE)
set(libmt32emu_CPP_INTERFACE ON CACHE BOOL "" FORCE)
set(libmt32emu_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(libmt32emu_WITH_INTERNAL_RESAMPLER ON CACHE BOOL "" FORCE)
set(libmt32emu_WITH_VERSION_TAGGING OFF CACHE BOOL "" FORCE)
set(libmt32emu_WITH_SYMBOL_VERSIONING OFF CACHE BOOL "" FORCE)
set(BUILD_SNAPSHOTS OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." mt32emu-core)
set_property(TARGET mt32emu PROPERTY VERSION)
set_property(TARGET mt32emu PROPERTY SOVERSION)
set_target_properties(mt32emu PROPERTIES OUTPUT_NAME mt32emu RUNTIME_OUTPUT_NAME mt32emu)
if(WIN32)
    set_target_properties(mt32emu PROPERTIES PREFIX "")
endif()
if(MINGW)
    target_link_options(mt32emu PRIVATE -static-libgcc -static-libstdc++)
endif()
if(APPLE)
    set_target_properties(mt32emu PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath")
endif()
